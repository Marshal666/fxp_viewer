cmake_minimum_required(VERSION 3.20)
project(fxp_viewer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(FXP_VIEWER_TOP_LEVEL ON)
else()
    set(FXP_VIEWER_TOP_LEVEL OFF)
endif()
option(FXP_VIEWER_BUILD_APP "Build desktop viewer" ${FXP_VIEWER_TOP_LEVEL})
option(FXP_VIEWER_BUILD_TESTS "Build tests" ${FXP_VIEWER_TOP_LEVEL})
option(FXP_VIEWER_BUILD_EXAMPLES "Build API example" ${FXP_VIEWER_TOP_LEVEL})
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
set(YAML_MSVC_SHARED_RT ON CACHE BOOL "" FORCE)
set(OPENGL_VERSION "3.3" CACHE STRING "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib URL https://codeload.github.com/raysan5/raylib/zip/02566453880e06bec97f5b8160fa6fe1e98397ca URL_HASH SHA256=999b52a128673948c6607d8cb841e34cc7ab7f13964ef2988dfce4fa7676b14b)
FetchContent_Declare(raygui URL https://codeload.github.com/raysan5/raygui/zip/020a61bebcbe288b4414de3416e219ef40af847a URL_HASH SHA256=3727f9803e3ed13bdd10d7a5a6b5b71d0bd2417fd5f8cfcaa4fa152336679da1)
FetchContent_Declare(yaml_cpp URL https://codeload.github.com/jbeder/yaml-cpp/zip/f7320141120f720aecc4c32be25586e7da9eb978 URL_HASH SHA256=af46f1da698244a3470f427756a83403aa378465f613ef5f09dec0f8cbb27cbc)
FetchContent_MakeAvailable(raylib yaml_cpp)
FetchContent_GetProperties(raygui)
if(NOT raygui_POPULATED)
  FetchContent_Populate(raygui)
endif()
if(NOT TARGET fxp::physics)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/modules/fxp_physics/external/fxp_math/CMakeLists.txt")
    message(FATAL_ERROR "Initialize physics dependencies: git submodule update --init --recursive")
  endif()
  add_subdirectory(modules/fxp_physics)
endif()
add_library(fxp_viewer_lib src/scene.cpp src/renderer.cpp src/scene_loader.cpp src/viewer.cpp src/gui.cpp src/physics_interactor.cpp)
add_library(fxp_viewer::viewer ALIAS fxp_viewer_lib)
target_include_directories(fxp_viewer_lib PUBLIC include PRIVATE src)
target_link_libraries(fxp_viewer_lib PUBLIC raylib fxp::physics PRIVATE yaml-cpp::yaml-cpp)
target_compile_features(fxp_viewer_lib PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(fxp_viewer_lib PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(fxp_viewer_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FXP_VIEWER_BUILD_APP)
  add_executable(fxp_viewer src/main.cpp)
  target_link_libraries(fxp_viewer PRIVATE fxp_viewer::viewer)
endif()
if(FXP_VIEWER_BUILD_EXAMPLES)
  add_executable(fxp_viewer_api_example examples/api_example.cpp)
  target_link_libraries(fxp_viewer_api_example PRIVATE fxp_viewer::viewer)
endif()
if(FXP_VIEWER_BUILD_TESTS)
  enable_testing()
  add_executable(fxp_viewer_tests tests/tests.cpp)
  target_link_libraries(fxp_viewer_tests PRIVATE fxp_viewer::viewer)
  add_test(NAME scene_api COMMAND fxp_viewer_tests)
  add_executable(fxp_viewer_physics_tests tests/physics_tests.cpp)
  target_link_libraries(fxp_viewer_physics_tests PRIVATE fxp_viewer::viewer)
  add_test(NAME physics_playback COMMAND fxp_viewer_physics_tests)
endif()

target_include_directories(fxp_viewer_lib SYSTEM PRIVATE
    "${raylib_SOURCE_DIR}/src" "${raylib_SOURCE_DIR}/src/external"
    "${raygui_SOURCE_DIR}/src")
if(MSVC AND FXP_VIEWER_BUILD_APP)
    set_target_properties(fxp_viewer PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_options(fxp_viewer PRIVATE /ENTRY:mainCRTStartup)
endif()
